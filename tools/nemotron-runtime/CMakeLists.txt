cmake_minimum_required(VERSION 3.22)
project(sound_nemotron_runtime LANGUAGES C CXX)
set(TRANSCRIBE_SOURCE "" CACHE PATH "Pinned transcribe.cpp checkout")
if(NOT EXISTS "${TRANSCRIBE_SOURCE}/include/transcribe.h")
    message(FATAL_ERROR "Provide pinned transcribe.cpp source")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_CUDA OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_VULKAN OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${TRANSCRIBE_SOURCE}" transcribe EXCLUDE_FROM_ALL)
target_include_directories(transcribe PUBLIC "${TRANSCRIBE_SOURCE}/include")
add_library(sound-nemotron-runtime SHARED bridge.cpp)
target_compile_features(sound-nemotron-runtime PRIVATE cxx_std_17)
target_include_directories(sound-nemotron-runtime PRIVATE "${TRANSCRIBE_SOURCE}/include")
target_link_libraries(sound-nemotron-runtime PRIVATE transcribe)
set_target_properties(sound-nemotron-runtime PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(sound-nemotron-runtime PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic-functions)
